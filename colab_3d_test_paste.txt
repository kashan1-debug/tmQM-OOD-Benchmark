def rmse(y, p):

def metrics(y, p):

# 2. Parse xyz frames for the sampled codes only (one streaming pass)

def parse_xyz_for_codes(data_dir, wanted):
    """Return {CSD_code: (elements, coords)} for wanted codes only."""
    files = sorted(glob.glob(os.path.join(data_dir, "tmQM_X*.xyz*")))
    print(f"Reading xyz from {len(files)} file(s)...")
    out = {}
    for f in files:
        opener = gzip.open if f.endswith(".gz") else open
        with opener(f, "rt") as fh:
            lines = fh
            while True:
                try:
                    first = next(lines)
                except StopIteration:
                    break
                first = first.strip()
                if not first.isdigit():
                    continue
                n = int(first)
                try:
                    comment = next(lines)
                except StopIteration:
                    break
                m = HEADER_RE.search(comment)
                code = m.group(1) if m else None
                atoms = []
                ok_frame = True
                for _ in range(n):
                    try:
                        parts = next(lines).split()
                    except StopIteration:
                        ok_frame = False
                        break
                    if len(parts) < 4:
                        ok_frame = False
                        break
                    atoms.append((parts[0], float(parts[1]), float(parts[2]), float(parts[3])))
                if ok_frame and code in wanted:
                    els = [a[0] for a in atoms]
                    xyz = np.array([[a[1], a[2], a[3]] for a in atoms], dtype=float)
                    out[code] = (els, xyz)
                    if len(out) % 1000 == 0:
                        print(f"  found {len(out):,}/{len(wanted):,} frames...")
                    if len(out) == len(wanted):
                        print(f"  all {len(wanted):,} frames found, stopping early.")
                        return out
    return out

# 3. 3D features from coordinates (metal = the one transition-metal atom)
# --------------------------------------------------------------------------- #
def d_electrons(symbol):
    z, g, p, r, en = METAL_PROPS[symbol]
    if symbol in ("La", "Ce", "Pr", "Nd", "Sm", "Eu", "Gd", "Tb", "Dy", "Ho", "Er", "Tm", "Yb", "Lu"):
        return 1
    return g - 2  # groups 3-12 -> d1-d10

def featurise_3d(els, xyz):
    """~30 interpretable 3D features. Never returns NaN (0.0 defaults)."""
    tm_idx = [i for i, e in enumerate(els) if e in METALS]
    if len(tm_idx) != 1:
        return None
    mi = tm_idx[0]
    metal = els[mi]
    z, g, p, r_cov, en_m = METAL_PROPS[metal]
    vec = xyz - xyz[mi]
    dist = np.sqrt((vec ** 2).sum(axis=1))
    dist[mi] = np.inf
    order = np.argsort(dist)
    d6 = dist[order[:6]]
    d6 = d6[np.isfinite(d6)]
    coord = [i for i in range(len(els)) if i != mi and dist[i] < 2.8]
    # --- bond lengths among coordinating atoms ---
    bd = dist[coord] if coord else np.array([0.0])
    # --- donor composition ---
    donors = [els[i] for i in coord]
    nN = donors.count("N"); nO = donors.count("O"); nC = donors.count("C")
    nS = donors.count("S"); nP = donors.count("P")
    nHal = sum(donors.count(h) for h in ("F", "Cl", "Br", "I"))
    nOther = len(donors) - (nN + nO + nC + nS + nP + nHal)
    en_d = float(np.mean([EN.get(e, 2.0) for e in donors])) if donors else 0.0
    rad_d = float(np.mean([RAD.get(e, 1.0) for e in donors])) if donors else 0.0
    # --- angles at the metal between coordinating atoms ---
    if len(coord) >= 2:
        v = vec[coord]
        v = v / np.sqrt((v ** 2).sum(axis=1, keepdims=True))
        angs = []
        for a in range(len(v)):
            for b in range(a + 1, len(v)):
                c = float(np.clip(np.dot(v[a], v[b]), -1.0, 1.0))
                angs.append(np.degrees(np.arccos(c)))
        angs = np.array(angs)
        mean_ang, std_ang = float(angs.mean()), float(angs.std())
        min_ang, max_ang = float(angs.min()), float(angs.max())
        frac90 = float(((angs >= 80) & (angs <= 100)).mean())
        frac180 = float(((angs >= 170) & (angs <= 180)).mean())
    else:
        mean_ang = std_ang = min_ang = max_ang = frac90 = frac180 = 0.0
    # --- size ---
    centroid = xyz.mean(axis=0)
    rg = float(np.sqrt(((xyz - centroid) ** 2).sum(axis=1).mean()))
    extent = float(np.sqrt((vec ** 2).sum(axis=1).max()))
    mean_bond = float(bd.mean()); std_bond = float(bd.std())
    feats = {
        "natoms": len(els), "cn_2.8": len(coord),
        "mean_d6": float(d6.mean()) if len(d6) else 0.0,
        "min_d6": float(d6.min()) if len(d6) else 0.0,
        "max_d6": float(d6.max()) if len(d6) else 0.0,
        "std_d6": float(d6.std()) if len(d6) else 0.0,
        "mean_bond": mean_bond, "std_bond": std_bond,
        "min_bond": float(bd.min()), "max_bond": float(bd.max()),
        "rel_spread": std_bond / mean_bond if mean_bond > 0 else 0.0,
        "n_N": nN, "n_O": nO, "n_C": nC, "n_S": nS, "n_P": nP, "n_hal": nHal, "n_other": nOther,
        "mean_donor_EN": en_d, "mean_donor_rad": rad_d,
        "mean_angle": mean_ang, "std_angle": std_ang, "min_angle": min_ang, "max_angle": max_ang,
        "frac_90": frac90, "frac_180": frac180,
        "Z_metal": z, "group_metal": g, "period_metal": p,
        "r_cov_metal": r_cov, "EN_metal": en_m, "d_electrons": d_electrons(metal),
        "rg": rg, "extent": extent,
    }
    return metal, feats

# 4. 2D baseline on the SAME rows: 256-bit Morgan FP (RDKit)

def make_morgan(n_bits=256, radius=2):
    try:
        from rdkit.Chem import rdFingerprintGenerator
        gen = rdFingerprintGenerator.GetMorganGenerator(radius=radius, fpSize=n_bits)
        def _fp(mol):
            arr = np.zeros((n_bits,), dtype=np.uint8)
            cDataStructs.ConvertToNumpyArray(gen.GetFingerprint(mol), arr)
            return arr
        return _fp
    except Exception:
        from rdkit.Chem import AllChem
        def _fp(mol):
            arr = np.zeros((n_bits,), dtype=np.uint8)
            cDataStructs.ConvertToNumpyArray(
                AllChem.GetMorganFingerprintAsBitVect(mol, radius, nBits=n_bits), arr)
            return arr
        return _fp

def grouped_folds(groups, seed, k):
    """Size-balanced greedy group -> fold assignment (same as v2 script)."""
    rng = np.random.default_rng(seed)
    uniq, counts = np.unique(groups, return_counts=True)
    sk = np.empty(len(uniq), dtype=[("count", int), ("rand", float)])
    sk["count"] = -counts; sk["rand"] = rng.random(len(uniq))
    order = np.argsort(sk, order=("count", "rand"))
    sizes = np.zeros(k, dtype=int); assign = {}
    for g, c in zip(uniq[order], counts[order]):
        f = int(np.argmin(sizes)); assign[g] = f; sizes[f] += c
    fold_of = np.array([assign[g] for g in groups])
    folds = [(np.where(fold_of != f)[0], np.where(fold_of == f)[0]) for f in range(k)]
    for tr, te in folds:
        assert set(groups[tr]).isdisjoint(set(groups[te])), "Group leakage!"
    return folds

def oof(factory, X, y, folds):
    yp = np.full(len(y), np.nan)
    for tr, te in folds:
        m = factory(); m.fit(X[tr], y[tr]); yp[te] = m.predict(X[te])
    return yp

    "XGBoost": lambda: XGBRegressor(n_estimators=200, learning_rate=0.05, max_depth=6, tree_method="hist", n_jobs=-1, random_state=0),
}

for split, folds in [("random", folds_random), ("grouped", folds_grouped)]:
    for feat, X in feature_sets.items():
        for name, fac in MODELS.items():
            yp = oof(fac, X, y, folds)
            r = metrics(y, yp)
            results.append({"Split": split, "Features": feat, "Model": name, **r})
            print(f"  {split:8s} | {feat:16s} | {name:13s} R2={r['R2']: .3f} RMSE={r['RMSE']:.3f} MAE={r['MAE']:.3f}")
